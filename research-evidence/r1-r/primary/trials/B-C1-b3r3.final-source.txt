/**
 * Migrate a legacy service configuration document to the v2 shape.
 *
 * Legacy documents are hand-written, so they spell the same setting in several ways
 * (`retry` / `retries` / `retryCount` / `attempts`, `timeout` / `timeout_ms` / `timeoutMs`, ...).
 *
 * The migration is a FAITHFUL READING of the original document, which fixes the order of work:
 *
 *   1. read the document itself  — which settings it names, under which spellings, and whether any
 *      single setting is named more than once (an ambiguous document has no faithful reading, so it
 *      must be refused rather than silently resolved by "first spelling wins");
 *   2. validate every value the document actually provides — a value that is present but wrong is an
 *      error, never something to paper over with a default;
 *   3. only then fill in the v2 defaults for settings the document genuinely omits.
 *
 * Doing defaults or coercion first erases the very information the ambiguity check needs.
 */

export class ConfigError extends Error {
  readonly code: string;
  constructor(code: string, message: string) {
    super(message);
    this.name = "ConfigError";
    this.code = code;
  }
}

export interface V2Config {
  readonly schemaVersion: 2;
  readonly service: { readonly name: string; readonly endpoint: string };
  readonly policy: { readonly retries: number; readonly timeoutMs: number };
}

const DEFAULT_RETRIES = 3;
const DEFAULT_TIMEOUT_MS = 30_000;

/** The v2 settings a legacy document may carry. */
type Setting = "name" | "endpoint" | "retries" | "timeoutMs";

/**
 * Spellings are compared only after normalization: hand-written keys differ in case and in the
 * separator they use (`retryCount`, `retry_count`, `retry-count`, `Retry Count` are one spelling).
 */
function normalizeSpelling(key: string): string {
  return key.toLowerCase().replace(/[^a-z0-9]/g, "");
}

const SETTINGS_BY_SPELLING: ReadonlyMap<string, Setting> = (() => {
  const table = new Map<string, Setting>();
  const add = (setting: Setting, ...spellings: string[]): void => {
    for (const spelling of spellings) {
      const normalized = normalizeSpelling(spelling);
      const existing = table.get(normalized);
      if (existing !== undefined && existing !== setting) {
        throw new Error(`internal: "${spelling}" is claimed by both ${existing} and ${setting}`);
      }
      table.set(normalized, setting);
    }
  };
  add("name", "name", "service_name", "serviceName", "service-name", "service name", "title");
  add("endpoint", "endpoint", "url", "base_url", "baseUrl", "base-url", "base url", "host");
  add("retries", "retry", "retries", "retry_count", "retryCount", "retry-count", "retry count", "attempts");
  add("timeoutMs", "timeout", "timeout_ms", "timeoutMs", "timeout-ms", "timeout ms");
  return table;
})();

/** Migrate a legacy configuration document to the v2 shape. */
export function migrateConfig(input: Record<string, unknown>): V2Config {
  if (input === null || typeof input !== "object" || Array.isArray(input)) {
    throw new ConfigError("INVALID_INPUT", "the configuration must be a plain object");
  }

  // Step 1 — read the document before transforming it.
  const spellingOf: Partial<Record<Setting, string>> = {};
  for (const key of Object.keys(input)) {
    const setting = SETTINGS_BY_SPELLING.get(normalizeSpelling(key));
    if (setting === undefined) {
      throw new ConfigError("UNKNOWN_SETTING", `"${key}" is not a legacy configuration setting`);
    }
    const previous = spellingOf[setting];
    if (previous !== undefined) {
      throw new ConfigError(
        "AMBIGUOUS_SETTING",
        `"${setting}" is given more than once, as "${previous}" and "${key}"`,
      );
    }
    spellingOf[setting] = key;
  }

  // Step 2 — validate present values. Step 3 — default only absent ones.
  const name = readName(input, spellingOf.name);
  const endpoint = readEndpoint(input, spellingOf.endpoint);
  const retries = readRetries(input, spellingOf.retries);
  const timeoutMs = readTimeoutMs(input, spellingOf.timeoutMs);

  return {
    schemaVersion: 2,
    service: { name, endpoint },
    policy: { retries, timeoutMs },
  };
}

function readName(input: Record<string, unknown>, key: string | undefined): string {
  if (key === undefined) throw new ConfigError("MISSING_NAME", "the service name is required");
  const value = input[key];
  if (typeof value !== "string") {
    throw new ConfigError("INVALID_NAME", "the service name must be a string");
  }
  if (value.trim().length === 0) {
    throw new ConfigError("INVALID_NAME", "the service name must not be empty or blank");
  }
  return value;
}

function readEndpoint(input: Record<string, unknown>, key: string | undefined): string {
  if (key === undefined) throw new ConfigError("MISSING_ENDPOINT", "the endpoint is required");
  const value = input[key];
  if (typeof value !== "string") {
    throw new ConfigError("INVALID_ENDPOINT", "the endpoint must be a string");
  }
  if (value.trim().length === 0) {
    throw new ConfigError("INVALID_ENDPOINT", "the endpoint must not be empty or blank");
  }
  return value;
}

function readRetries(input: Record<string, unknown>, key: string | undefined): number {
  if (key === undefined) return DEFAULT_RETRIES;
  const value = input[key];
  if (typeof value !== "number" || !Number.isInteger(value) || value < 0) {
    throw new ConfigError("INVALID_RETRIES", "retries must be a non-negative integer");
  }
  return value;
}

function readTimeoutMs(input: Record<string, unknown>, key: string | undefined): number {
  if (key === undefined) return DEFAULT_TIMEOUT_MS;
  const value = input[key];
  if (typeof value !== "number" || !Number.isInteger(value) || value <= 0) {
    throw new ConfigError("INVALID_TIMEOUT", "the timeout must be a positive integer number of milliseconds");
  }
  return value;
}
