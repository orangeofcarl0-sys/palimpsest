/**
 * Legacy -> v2 configuration migration.
 *
 * THE METHOD (this is the part that is easy to get wrong):
 *
 *   1. NORMALIZE the key spellings first. A legacy document is hand-written and the same setting is
 *      spelled several ways ("service_name", "serviceName" and "SERVICE-NAME" are one setting; so are
 *      "timeout_ms" and "timeoutMs"). Case and separators are not significant, only the canonical
 *      spelling is.
 *   2. VALIDATE the whole document before it is transformed or defaulted. A value that is present but
 *      unusable is an error, never a reason to quietly substitute a default: silently defaulting an
 *      operator's "retry: -1" (or "retry: many") changes what their document means.
 *   3. REFUSE AMBIGUITY. When one setting is given under more than one of its spellings, the values are
 *      compared; if they disagree the document is ambiguous and there is no faithful reading of it, so
 *      it is rejected instead of letting one spelling win by accident of ordering.
 *   4. REFUSE UNKNOWN KEYS. A key the migration does not understand would be dropped, and a dropped key
 *      is lost meaning. The legacy schema is closed; anything outside it is rejected.
 *
 * The governing invariant is faithfulness: a document that is accepted must mean *exactly* what it
 * meant before the migration. Anything the migration cannot read faithfully is refused.
 */

export class ConfigError extends Error {
  readonly code: string;
  constructor(code: string, message: string) {
    super(message);
    this.name = "ConfigError";
    this.code = code;
  }
}

export interface V2Config {
  readonly schemaVersion: 2;
  readonly service: { readonly name: string; readonly endpoint: string };
  readonly policy: { readonly retries: number; readonly timeoutMs: number };
}

/** Defaults for the settings the legacy format allowed an operator to leave out. */
const DEFAULT_RETRIES = 3;
const DEFAULT_TIMEOUT_MS = 30_000;

type Setting = "name" | "endpoint" | "retries" | "timeoutMs";

/**
 * The documented legacy spellings, keyed by their NORMALISED form (lower-case, separators removed).
 * Normalising is what makes "timeout_ms", "timeoutMs", "Timeout-MS" and "timeout ms" the same setting
 * *before* their values are compared.
 *
 * The keys of a legacy document are closed: a key that normalises to anything not listed here is an
 * unrecognised setting and the document is refused (see rule 4).
 */
const SETTING_OF_SPELLING: Readonly<Record<string, Setting>> = {
  name: "name",
  servicename: "name",
  title: "name",

  endpoint: "endpoint",
  url: "endpoint",
  baseurl: "endpoint",
  host: "endpoint",

  retry: "retries",
  retries: "retries",
  retrycount: "retries",
  attempts: "retries",

  timeout: "timeoutMs",
  timeoutms: "timeoutMs",
};

const SETTINGS: readonly Setting[] = ["name", "endpoint", "retries", "timeoutMs"];

/** Canonical form of a key spelling: case and separators carry no meaning in the legacy format. */
function normalizeSpelling(key: string): string {
  return key.toLowerCase().replace(/[^a-z0-9]/g, "");
}

function isPlainObject(value: unknown): value is Record<string, unknown> {
  if (value === null || typeof value !== "object" || Array.isArray(value)) return false;
  const prototype = Object.getPrototypeOf(value);
  return prototype === null || prototype === Object.prototype;
}

/**
 * Validate ONE value for a setting and return it in canonical form. Validation happens on the raw
 * legacy value: nothing is coerced and nothing is defaulted here, so an unusable value is reported
 * rather than masked.
 */
function readValue(setting: Setting, spelling: string, raw: unknown): string | number {
  switch (setting) {
    case "name":
    case "endpoint": {
      // A name or endpoint that is not a string, or is empty/blank, cannot be read faithfully.
      if (typeof raw !== "string" || raw.trim() === "") {
        throw new ConfigError(
          setting === "name" ? "INVALID_NAME" : "INVALID_ENDPOINT",
          `${spelling} must be a non-empty string`,
        );
      }
      return raw;
    }
    case "retries": {
      // Retries are a whole count and may legitimately be 0; negative or fractional counts may not.
      if (typeof raw !== "number" || !Number.isInteger(raw) || raw < 0) {
        throw new ConfigError("INVALID_RETRIES", `${spelling} must be a non-negative integer`);
      }
      return raw;
    }
    case "timeoutMs": {
      // A timeout of zero (or less) means "no timeout" in neither the old nor the new format.
      if (typeof raw !== "number" || !Number.isInteger(raw) || raw <= 0) {
        throw new ConfigError("INVALID_TIMEOUT", `${spelling} must be a positive integer number of milliseconds`);
      }
      return raw;
    }
  }
}

/** Migrate a legacy configuration document to the v2 shape, or refuse it. */
export function migrateConfig(input: Record<string, unknown>): V2Config {
  if (!isPlainObject(input)) {
    throw new ConfigError("INVALID_INPUT", "the configuration must be a plain object");
  }

  // Step 1 - normalise every key spelling and group the document's entries by the setting they name.
  const entriesBySetting = new Map<Setting, { spelling: string; value: unknown }[]>();
  for (const key of Object.keys(input)) {
    const setting = SETTING_OF_SPELLING[normalizeSpelling(key)];
    if (setting === undefined) {
      throw new ConfigError("UNKNOWN_KEY", `unrecognised legacy key ${JSON.stringify(key)}`);
    }
    const entries = entriesBySetting.get(setting);
    if (entries === undefined) entriesBySetting.set(setting, [{ spelling: key, value: input[key] }]);
    else entries.push({ spelling: key, value: input[key] });
  }

  // Step 2/3 - validate every value that is present, then refuse to choose between spellings that
  // disagree. Both happen before any default is applied.
  const resolved = new Map<Setting, string | number>();
  for (const setting of SETTINGS) {
    const entries = entriesBySetting.get(setting);
    if (entries === undefined) continue;

    const [first, ...rest] = entries;
    const value = readValue(setting, first.spelling, first.value);
    for (const entry of rest) {
      const other = readValue(setting, entry.spelling, entry.value);
      if (other !== value) {
        throw new ConfigError(
          "AMBIGUOUS_CONFIG",
          `the document gives ${setting} more than once and the spellings disagree: ` +
            `${first.spelling}=${JSON.stringify(first.value)} vs ${entry.spelling}=${JSON.stringify(entry.value)}`,
        );
      }
    }
    resolved.set(setting, value);
  }

  // Step 4 - required settings must be present; the rest fall back to the v2 defaults, now that the
  // document has been read and validated in full.
  const name = resolved.get("name");
  if (name === undefined) throw new ConfigError("MISSING_NAME", "the service name is required");

  const endpoint = resolved.get("endpoint");
  if (endpoint === undefined) throw new ConfigError("MISSING_ENDPOINT", "the endpoint is required");

  const retries = resolved.get("retries") ?? DEFAULT_RETRIES;
  const timeoutMs = resolved.get("timeoutMs") ?? DEFAULT_TIMEOUT_MS;

  return {
    schemaVersion: 2,
    service: { name: name as string, endpoint: endpoint as string },
    policy: { retries: retries as number, timeoutMs: timeoutMs as number },
  };
}
