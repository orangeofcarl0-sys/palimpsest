/**
 * Legacy → v2 configuration migration.
 *
 * The legacy format is hand-written, so the same setting is spelled several ways
 * (`service_name`/`serviceName`/`Service-Name`, `base_url`/`baseUrl`, `retry`/`retries`/
 * `retry_count`/`attempts`, `timeout`/`timeout_ms`/`timeoutMs`). A migrated document has to
 * mean exactly what the operator's document meant, which fixes the order of the work:
 *
 *   1. normalize the key spellings and refuse anything the legacy format does not define —
 *      a key we cannot place is a key we cannot honestly drop;
 *   2. resolve a setting that is named more than once by comparing the values its spellings
 *      carry: spellings that disagree leave the document ambiguous and it is refused, while
 *      spellings that agree are a redundant but unambiguous statement;
 *   3. validate the values the document actually states;
 *   4. only then fill in the v2 defaults, and only for settings the document never mentions.
 *
 * Steps 3 and 4 in that order are the point of the migration: a default must never paper
 * over a value the operator did write (a legacy `retry: "5"` is a stated retry count that we
 * cannot read, not a missing one).
 */

export class ConfigError extends Error {
  readonly code: string;
  constructor(code: string, message: string) {
    super(message);
    this.name = "ConfigError";
    this.code = code;
  }
}

export interface V2Config {
  readonly schemaVersion: 2;
  readonly service: { readonly name: string; readonly endpoint: string };
  readonly policy: { readonly retries: number; readonly timeoutMs: number };
}

/** The v2 settings a legacy document can speak about. */
type Setting = "name" | "endpoint" | "retries" | "timeout";

/** One setting as the document stated it: the spelling used, and the value it carried. */
interface Statement {
  readonly spelling: string;
  readonly value: unknown;
}

const DEFAULT_RETRIES = 3;
const DEFAULT_TIMEOUT_MS = 30_000;

/**
 * Every spelling the legacy format may use for a setting, written in normalized form
 * (see {@link normalizeSpelling}). The conventions are only partly documented, so a key is
 * matched by the setting it denotes rather than by literal text: case, underscores, dashes
 * and spaces are spelling, not meaning.
 */
const SPELLINGS: ReadonlyArray<readonly [Setting, readonly string[]]> = [
  ["name", ["name", "servicename", "title"]],
  ["endpoint", ["endpoint", "url", "baseurl", "host"]],
  ["retries", ["retry", "retries", "retrycount", "attempts"]],
  ["timeout", ["timeout", "timeoutms"]],
];

const SETTING_OF_SPELLING = new Map<string, Setting>();
for (const [setting, spellings] of SPELLINGS) {
  for (const spelling of spellings) SETTING_OF_SPELLING.set(spelling, setting);
}

/** `service-name`, `service_name`, `serviceName` and `Service Name` are one spelling. */
function normalizeSpelling(key: string): string {
  return key.toLowerCase().replace(/[^a-z0-9]+/g, "");
}

/** Structural equality for the JSON values a configuration document can carry. */
function sameValue(left: unknown, right: unknown): boolean {
  if (left === right) return true;
  if (left === null || right === null) return false;
  if (typeof left !== "object" || typeof right !== "object") return false;
  return stableStringify(left) === stableStringify(right);
}

function stableStringify(value: unknown): string {
  if (Array.isArray(value)) return `[${value.map(stableStringify).join(",")}]`;
  if (value !== null && typeof value === "object") {
    const record = value as Record<string, unknown>;
    const body = Object.keys(record)
      .sort()
      .map((key) => `${JSON.stringify(key)}:${stableStringify(record[key])}`)
      .join(",");
    return `{${body}}`;
  }
  return JSON.stringify(value) ?? "null";
}

/** Migrate a legacy configuration document to the v2 shape. */
export function migrateConfig(input: Record<string, unknown>): V2Config {
  if (input === null || typeof input !== "object" || Array.isArray(input)) {
    throw new ConfigError("INVALID_INPUT", "the configuration must be a plain object");
  }

  // (1) and (2): read the document the way it was written. Normalize every spelling, refuse
  // keys the legacy format does not define, and refuse a setting that is named twice with
  // values that disagree — such a document has no single faithful reading.
  const stated = new Map<Setting, Statement>();
  for (const spelling of Object.keys(input)) {
    const setting = SETTING_OF_SPELLING.get(normalizeSpelling(spelling));
    if (setting === undefined) {
      throw new ConfigError("UNKNOWN_KEY", `"${spelling}" is not a setting the legacy configuration format defines`);
    }
    const value = input[spelling];
    const earlier = stated.get(setting);
    if (earlier === undefined) {
      stated.set(setting, { spelling, value });
      continue;
    }
    if (!sameValue(earlier.value, value)) {
      throw new ConfigError(
        "AMBIGUOUS_SETTING",
        `"${earlier.spelling}" and "${spelling}" both name the "${setting}" setting but disagree`,
      );
    }
  }

  // (3): validate the values the document actually states, before any default exists.
  const name = readString(stated, "name");
  const endpoint = readString(stated, "endpoint");
  const retries = readRetries(stated);
  const timeoutMs = readTimeoutMs(stated);

  if (name === undefined) throw new ConfigError("INVALID_NAME", "the service name must be a non-empty string");
  if (endpoint === undefined) throw new ConfigError("INVALID_ENDPOINT", "the endpoint must be a non-empty string");

  // (4): defaults last, and only for the settings the document never mentions.
  return {
    schemaVersion: 2,
    service: { name, endpoint },
    policy: {
      retries: retries ?? DEFAULT_RETRIES,
      timeoutMs: timeoutMs ?? DEFAULT_TIMEOUT_MS,
    },
  };
}

function readString(stated: Map<Setting, Statement>, setting: "name" | "endpoint"): string | undefined {
  const statement = stated.get(setting);
  if (statement === undefined) return undefined;
  const value = statement.value;
  if (typeof value !== "string" || value.trim() === "") {
    const code = setting === "name" ? "INVALID_NAME" : "INVALID_ENDPOINT";
    throw new ConfigError(code, `"${statement.spelling}" must be a non-empty string`);
  }
  return value;
}

function readRetries(stated: Map<Setting, Statement>): number | undefined {
  const statement = stated.get("retries");
  if (statement === undefined) return undefined;
  const value = statement.value;
  if (typeof value !== "number" || !Number.isInteger(value) || value < 0) {
    throw new ConfigError("INVALID_RETRIES", `"${statement.spelling}" must be a whole number of retries, 0 or more`);
  }
  return value;
}

function readTimeoutMs(stated: Map<Setting, Statement>): number | undefined {
  const statement = stated.get("timeout");
  if (statement === undefined) return undefined;
  const value = statement.value;
  if (typeof value !== "number" || !Number.isFinite(value) || value <= 0) {
    throw new ConfigError("INVALID_TIMEOUT", `"${statement.spelling}" must be a positive number of milliseconds`);
  }
  return value;
}
